cmake_minimum_required(VERSION 3.20)
project(PaperView VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  add_compile_options(/utf-8)
endif()
add_library(paper_core src/core.cpp src/settings.cpp)
target_include_directories(paper_core PUBLIC src)
if(WIN32)
  enable_language(RC)
  add_executable(PaperView WIN32 src/main.cpp src/app.cpp src/backend.cpp src/renderer.cpp src/toolbar.cpp src/settings_dialog.cpp src/office.cpp src/office_convert.cpp packaging/app.rc)
  target_include_directories(PaperView PRIVATE packaging)
  target_compile_definitions(PaperView PRIVATE UNICODE _UNICODE NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
  target_link_libraries(PaperView PRIVATE paper_core d2d1 dwrite comctl32 comdlg32 shell32 ole32 uuid uxtheme bcrypt gdi32 user32 oleaut32 advapi32)
  if(MINGW)
    target_compile_options(PaperView PRIVATE -Wall -Wextra -Wno-missing-field-initializers -Wno-misleading-indentation -Wno-cast-function-type -Wno-unknown-pragmas)
    target_link_options(PaperView PRIVATE -municode -static -static-libgcc -static-libstdc++)
  elseif(MSVC)
    target_compile_definitions(PaperView PRIVATE PAPERVIEW_EXTERNAL_MANIFEST=1)
    target_sources(PaperView PRIVATE packaging/app.manifest)
    target_compile_options(PaperView PRIVATE /utf-8 /W4)
  endif()
endif()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE paper_core)
if(MINGW)
  target_link_options(core_tests PRIVATE -static -static-libgcc -static-libstdc++)
endif()
add_test(NAME core_tests COMMAND core_tests)

add_executable(settings_tests tests/settings_tests.cpp)
target_link_libraries(settings_tests PRIVATE paper_core)
if(MINGW)
  target_link_options(settings_tests PRIVATE -static -static-libgcc -static-libstdc++)
endif()
add_test(NAME settings_tests COMMAND settings_tests)

if(WIN32)
  add_executable(native_office_smoke tests/native_office_smoke.cpp src/office.cpp src/backend.cpp)
  target_include_directories(native_office_smoke PRIVATE src)
  target_compile_definitions(native_office_smoke PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(native_office_smoke PRIVATE paper_core ole32 oleaut32 uuid user32 gdi32 advapi32 bcrypt shell32)
  if(MINGW)
    target_link_options(native_office_smoke PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_executable(settings_ui_smoke tests/settings_ui_smoke.cpp src/settings_dialog.cpp)
  target_include_directories(settings_ui_smoke PRIVATE src)
  target_compile_definitions(settings_ui_smoke PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(settings_ui_smoke PRIVATE paper_core comctl32 user32 gdi32)
  if(MINGW)
    target_link_options(settings_ui_smoke PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME settings_ui_smoke COMMAND settings_ui_smoke)
  set_tests_properties(settings_ui_smoke PROPERTIES TIMEOUT 30)
  add_test(NAME native_office_smoke COMMAND native_office_smoke)
endif()
