cmake_minimum_required(VERSION 3.20)
project(PaperView VERSION 0.3.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(paper_core src/core.cpp)
target_include_directories(paper_core PUBLIC src)
if(WIN32)
  enable_language(RC)
  add_executable(PaperView WIN32 src/main.cpp src/app.cpp src/backend.cpp src/renderer.cpp src/toolbar.cpp packaging/app.rc)
  target_compile_definitions(PaperView PRIVATE UNICODE _UNICODE NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
  target_link_libraries(PaperView PRIVATE paper_core d2d1 dwrite comctl32 comdlg32 shell32 ole32 uuid uxtheme bcrypt gdi32 user32)
  if(MINGW)
    target_compile_options(PaperView PRIVATE -Wall -Wextra -Wno-missing-field-initializers -Wno-misleading-indentation -Wno-cast-function-type -Wno-unknown-pragmas)
    target_link_options(PaperView PRIVATE -municode -static -static-libgcc -static-libstdc++)
  elseif(MSVC)
    target_compile_options(PaperView PRIVATE /utf-8 /W4)
  endif()
endif()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE paper_core)
if(MINGW)
  target_link_options(core_tests PRIVATE -static -static-libgcc -static-libstdc++)
endif()
add_test(NAME core_tests COMMAND core_tests)
